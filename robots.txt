# robots.txt for votega.org
#
# Policy: search engines and user-initiated AI retrieval are welcome.
# Crawling to build AI *training* corpora is not.
#
# Several AI companies run separate crawlers for separate purposes. Blocking a
# training crawler does not block that company's search crawler, so the rules
# below refuse training while keeping VoteGA citable in AI search results —
# which is a source of traffic, not a leak of it.
#
# NOTE: robots.txt is advisory. Major operators honor it; bad actors ignore it.
# Enforcement requires blocking at the edge (Cloudflare AI Crawl Control / WAF).

# Cloudflare/IETF content-signal declaration. Advisory, mirrors the rules below.
# search    = may index for search results
# ai-input  = may retrieve at query time to answer a user and cite the source
# ai-train  = may use to train or fine-tune a model
Content-Signal: search=yes, ai-input=yes, ai-train=no

User-agent: *
Allow: /
